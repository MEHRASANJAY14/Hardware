function saveHardware() { try { var solId = document.getElementById( "solId" ).value; var branchName = document.getElementById( "branchName" ).value; if (!solId) { alert( "Please select SOL ID." ); return; } if (!branchName) { alert( "Branch Name not found." ); return; } var cards = document.querySelectorAll( ".hardware-item" ); if ( cards.length === 0 ) { alert( "Please add hardware." ); return; } var data = []; for ( var i = 0; i < cards.length; i++ ) { var card = cards[i]; var hardwareType = card.querySelector( ".hardwareType" ).value; var hardwareSubType = card.querySelector( ".hardwareSubType" ).value; var make = card.querySelector( ".make" ).value .trim(); var model = card.querySelector( ".model" ).value .trim(); var serialNo = card.querySelector( ".serialNo" ).value .trim(); var condition = card.querySelector( ".condition" ).value; var remarks = card.querySelector( ".remarks" ).value .trim(); if (!hardwareType) { alert( "Please select Hardware Type for item #" + (i + 1) ); return; } if (!make) { alert( "Please enter Make for item #" + (i + 1) ); return; } if (!model) { alert( "Please enter Model for item #" + (i + 1) ); return; } if (!serialNo) { alert( "Please enter Serial No for item #" + (i + 1) ); return; } if (!condition) { alert( "Please select Condition for item #" + (i + 1) ); return; } data.push({ solId: String(solId), branchName: String(branchName), hardwareType: String(hardwareType), hardwareSubType: String( hardwareSubType || "" ), make: String(make), model: String(model), serialNo: String(serialNo), condition: String(condition), remarks: String(remarks || "") }); } /* * Convert the array to a simple JSON string. * This is the important fix. */ var jsonData = JSON.stringify( data ); showLoading( "Saving hardware..." ); google.script.run .withSuccessHandler( function(result) { hideLoading(); if ( result && result.status === "success" ) { alert( "✅ Hardware Saved Successfully\n\n" + "Items: " + result.count + "\nSubmission ID: " + result.submissionId ); resetHardwareForm(); /* * Refresh dashboard */ loadDashboard(); } else { alert( "❌ Save Failed\n\n" + ( result && result.message ? result.message : "Unknown error" ) ); } } ) .withFailureHandler( function(error) { hideLoading(); alert( "❌ Server Error\n\n" + ( error && error.message ? error.message : String(error) ) ); } ) .saveHardwareBatch( jsonData ); } catch (error) { hideLoading(); alert( "❌ Browser Error\n\n" + ( error && error.message ? error.message : String(error) ) ); } }
